```sql
-- Cola de notificaciones
create table notification_queue (
  id bigserial primary key,
  bidder_session_id text not null,
  lot_id bigint not null references lots(id),
  channel text not null check (channel in ('email','whatsapp')),
  recipient text not null,
  subject text,
  body text not null,
  status text default 'pending' check (status in ('pending','sent','failed')),
  attempts int default 0,
  error text,
  created_at timestamptz default now(),
  sent_at timestamptz
);

create index idx_queue_pending on notification_queue(status, created_at)
  where status = 'pending';

-- Contactos del bidder (email/whatsapp)
alter table bidders add column if not exists email text;
alter table bidders add column if not exists whatsapp text;
alter table bidders add column if not exists notify_on_outbid boolean default true;

-- Trigger: cuando alguien supera tu puja, encolar notificación
create or replace function enqueue_outbid_notification()
returns trigger as $$
declare
  prev_bid record;
  new_leader record;
begin
  -- Buscar la puja inmediatamente anterior (del nuevo líder)
  select * into new_leader from bids
    where lot_id = NEW.lot_id and id <> NEW.id
    order by amount desc, created_at desc limit 1;

  if new_leader is null then return NEW; end if;

  -- Buscar datos del bidder que fue superado
  insert into notification_queue (bidder_session_id, lot_id, channel, recipient, subject, body)
  select
    b.session_id,
    NEW.lot_id,
    'email',
    bd.email,
    '⚠️ Te superaron en el lote ' || l.title,
    'Hola ' || bd.name || ', tu puja de US$ ' || new_leader.amount ||
    ' en el lote "' || l.title || '" fue superada. Nueva puja: US$ ' || NEW.amount ||
    '. Entrá a pujar de nuevo antes del cierre.'
  from bids b
  join bidders bd on bd.session_id = b.bidder_session_id
  join lots l on l.id = NEW.lot_id
  where b.id = new_leader.id
    and bd.email is not null
    and bd.notify_on_outbid = true
    and bd.session_id <> NEW.bidder_session_id;  -- no notificarte a vos mismo

  -- Lo mismo para WhatsApp
  insert into notification_queue (bidder_session_id, lot_id, channel, recipient, subject, body)
  select
    b.session_id,
    NEW.lot_id,
    'whatsapp',
    bd.whatsapp,
    null,
    '🚜 *Remate #59*%0A%0ATe superaron en el lote *' || l.title || '*%0A' ||
    'Tu puja: US$ ' || new_leader.amount || '%0A' ||
    'Nueva puja: *US$ ' || NEW.amount || '*%0A%0A' ||
    'Pujá de nuevo: ' || coalesce(current_setting('app.site_url', true), 'https://remates-agricolas.com.uy') || '/lote/' || l.id
  from bids b
  join bidders bd on bd.session_id = b.bidder_session_id
  join lots l on l.id = NEW.lot_id
  where b.id = new_leader.id
    and bd.whatsapp is not null
    and bd.notify_on_outbid = true
    and bd.session_id <> NEW.bidder_session_id;

  return NEW;
end;
$$ language plpgsql;

drop trigger if exists trg_enqueue_outbid on bids;
create trigger trg_enqueue_outbid
  after insert on bids
  for each row execute function enqueue_outbid_notification();
```

```typescript
import { Resend } from 'resend'

const resend = new Resend(process.env.RESEND_API_KEY)

export async function sendEmail(to: string, subject: string, body: string) {
  if (!process.env.RESEND_API_KEY) {
    console.log('[DEV] Email no enviado:', { to, subject, body })
    return { ok: true, dev: true }
  }

  const res = await resend.emails.send({
    from: 'Remates Agrícolas <no-reply@remates-agricolas.com.uy>',
    to,
    subject,
    text: body,
  })

  if (res.error) return { ok: false, error: res.error.message }
  return { ok: true, id: res.data?.id }
}
```

### `lib/notifications/twilio.ts`

```typescript
import twilio from 'twilio'

const client = process.env.TWILIO_ACCOUNT_SID
  ? twilio(process.env.TWILIO_ACCOUNT_SID, process.env.TWILIO_AUTH_TOKEN)
  : null

export async function sendWhatsApp(to: string, body: string) {
  // Normalizar número: quitar + y espacios, agregar prefijo whatsapp:
  const phone = `whatsapp:+${to.replace(/\D/g, '')}`

  if (!client) {
    console.log('[DEV] WhatsApp no enviado:', { to, body })
    return { ok: true, dev: true }
  }

  try {
    const msg = await client.messages.create({
      from: `whatsapp:${process.env.TWILIO_WHATSAPP_NUMBER}`,
      to: phone,
      body: body.replace(/%0A/g, '\n').replace(/\*/g, ''),
    })
    return { ok: true, sid: msg.sid }
  } catch (err: any) {
    return { ok: false, error: err.message }
  }
}
```

### `app/api/cron/send-notifications/route.ts`

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { createServerSupabase } from '@/lib/supabase/server'
import { sendEmail } from '@/lib/notifications/resend'
import { sendWhatsApp } from '@/lib/notifications/twilio'

export const dynamic = 'force-dynamic'
export const maxDuration = 30

export async function GET(req: NextRequest) {
  const auth = req.headers.get('authorization')
  if (auth !== `Bearer ${process.env.CRON_SECRET}`) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  }

  const supabase = await createServerSupabase()
  const { data: pending } = await supabase
    .from('notification_queue')
    .select('*')
    .eq('status', 'pending')
    .lt('attempts', 3)
    .order('created_at')
    .limit(50)

  let sent = 0, failed = 0

  for (const n of pending ?? []) {
    try {
      const result = n.channel === 'email'
        ? await sendEmail(n.recipient, n.subject!, n.body)
        : await sendWhatsApp(n.recipient, n.body)

      if (result.ok) {
        await supabase.from('notification_queue').update({
          status: 'sent', sent_at: new Date().toISOString(),
          attempts: n.attempts + 1,
        }).eq('id', n.id)
        sent++
      } else {
        throw new Error(result.error)
      }
    } catch (err: any) {
      await supabase.from('notification_queue').update({
        status: n.attempts >= 2 ? 'failed' : 'pending',
        attempts: n.attempts + 1,
        error: err.message,
      }).eq('id', n.id)
      failed++
    }
  }

  return NextResponse.json({ processed: (pending ?? []).length, sent, failed })
}
```

```tsx
'use client'
import { useEffect, useState } from 'react'
import { Download, X } from 'lucide-react'

interface BeforeInstallPromptEvent extends Event {
  prompt: () => Promise<void>
  userChoice: Promise<{ outcome: 'accepted' | 'dismissed' }>
}

export function InstallPWAButton() {
  const [deferredPrompt, setDeferredPrompt] = useState<BeforeInstallPromptEvent | null>(null)
  const [show, setShow] = useState(false)
  const [isIOS, setIsIOS] = useState(false)

  useEffect(() => {
    const isIOSDevice = /iPad|iPhone|iPod/.test(navigator.userAgent) && !(window as any).MSStream
    const isStandalone = window.matchMedia('(display-mode: standalone)').matches
    const dismissed = localStorage.getItem('pwa_dismissed')

    if (isStandalone || dismissed) return

    if (isIOSDevice) {
      setIsIOS(true)
      setShow(true)
      return
    }

    const handler = (e: Event) => {
      e.preventDefault()
      setDeferredPrompt(e as BeforeInstallPromptEvent)
      setShow(true)
    }
    window.addEventListener('beforeinstallprompt', handler)
    return () => window.removeEventListener('beforeinstallprompt', handler)
  }, [])

  const handleInstall = async () => {
    if (!deferredPrompt) return
    deferredPrompt.prompt()
    const { outcome } = await deferredPrompt.userChoice
    if (outcome === 'accepted') setShow(false)
    setDeferredPrompt(null)
  }

  const dismiss = () => {
    setShow(false)
    localStorage.setItem('pwa_dismissed', '1')
  }

  if (!show) return null

  return (
    <div className="fixed bottom-20 left-4 right-4 md:left-auto md:right-4 md:w-80 z-50 animate-in slide-in-from-bottom-4">
      <div className="bg-white rounded-2xl shadow-2xl border-2 border-forest p-4 relative">
        <button
          onClick={dismiss}
          className="absolute top-2 right-2 w-6 h-6 rounded-full hover:bg-cream-dark flex items-center justify-center"
        >
          <X className="w-3 h-3" />
        </button>

        <div className="flex items-start gap-3">
          <div className="w-12 h-12 bg-forest rounded-xl flex items-center justify-center flex-shrink-0">
            <span className="text-2xl">🚜</span>
          </div>
          <div className="flex-1 min-w-0">
            <h3 className="font-display font-bold text-forest-dark">Instalá la app</h3>
            <p className="text-xs text-earth-soft mt-0.5">
              {isIOS
                ? 'Tocá el ícono de compartir ↑ y luego "Agregar a inicio"'
                : 'Recibí alertas de pujas al instante, incluso sin abrir el navegador'}
            </p>
            {!isIOS && (
              <button
                onClick={handleInstall}
                className="mt-2 flex items-center gap-1.5 px-3 py-1.5 bg-forest hover:bg-forest-dark text-white text-xs font-semibold rounded-lg"
              >
                <Download className="w-3.5 h-3.5" />
                Instalar
              </button>
            )}
          </div>
        </div>
      </div>
    </div>
  )
}
```

### `supabase/migrations/analytics.sql`

```sql
create table page_views (
  id bigserial primary key,
  path text not null,
  lot_id bigint,
  session_id text not null,
  referrer text,
  user_agent text,
  country text,
  created_at timestamptz default now()
);

create index idx_pv_lot on page_views(lot_id, created_at desc);
create index idx_pv_created on page_views(created_at desc);

-- Resumen materializado (se actualiza cada hora vía pg_cron o Vercel Cron)
create materialized view lot_stats as
select
  l.id as lot_id,
  l.title,
  l.album,
  l.starting_price,
  l.status,
  (select count(*) from page_views pv where pv.lot_id = l.id) as total_views,
  (select count(*) from bids b where b.lot_id = l.id) as total_bids,
  (select count(distinct bidder_session_id) from bids where lot_id = l.id) as unique_bidders,
  (select max(amount) from bids where lot_id = l.id) as highest_bid,
  case
    when (select count(*) from bids where lot_id = l.id) > 0 then 1
    else 0
  end as converted
from lots l;

create unique index idx_lot_stats_id on lot_stats(lot_id);

-- Función para refrescar
create or replace function refresh_lot_stats()
returns void as $$
begin
  refresh materialized view concurrently lot_stats;
end;
$$ language plpgsql;
```

### `middleware.ts` (trackear page views)

```typescript
import { NextResponse } from 'next/server'
import type { NextRequest } from 'next/server'

export async function middleware(req: NextRequest) {
  const res = NextResponse.next()

  // Trackear solo páginas de lote
  const match = req.nextUrl.pathname.match(/^\/lote\/(\d+)$/)
  if (match && req.method === 'GET') {
    const lotId = Number(match[1])
    const sessionId = req.cookies.get('bidder_sid')?.value || req.headers.get('x-forwarded-for') || 'anon'

    // Fire-and-forget (no bloquea la respuesta)
    fetch(`${req.nextUrl.origin}/api/analytics/track`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        path: req.nextUrl.pathname,
        lotId,
        sessionId,
        referrer: req.headers.get('referer'),
        userAgent: req.headers.get('user-agent'),
        country: req.headers.get('x-vercel-ip-country'),
      }),
    }).catch(() => {})
  }

  return res
}

export const config = { matcher: ['/lote/:path*'] }
```

### `supabase/functions/analytics.sql` (funciones agregadas)

```sql
create or replace function get_global_stats()
returns jsonb as $$
declare
  result jsonb;
begin
  select jsonb_build_object(
    'total_views', (select count(*) from page_views where created_at > now() - interval '7 days'),
    'unique_visitors', (select count(distinct session_id) from page_views where created_at > now() - interval '7 days'),
    'total_bids', (select count(*) from bids where created_at > now() - interval '7 days'),
    'active_bidders', (select count(distinct bidder_session_id) from bids where created_at > now() - interval '7 days'),
    'revenue_potential', (select coalesce(sum(max_bid), 0) from (
      select max(amount) as max_bid from bids group by lot_id
    ) as sub),
    'conversion_rate', (
      select round(100.0 * count(*) filter (where converted = 1) / nullif(count(*), 0), 1)
      from lot_stats
    )
  ) into result;
  return result;
end;
$$ language plpgsql;

create or replace function get_conversion_funnel()
returns jsonb as $$
declare
  result jsonb;
begin
  select jsonb_build_object(
    'lot_views', (select count(*) from page_views),
    'unique_viewers', (select count(distinct session_id) from page_views),
    'bidders', (select count(distinct bidder_session_id) from bids),
    'total_bids', (select count(*) from bids),
    'won_lots', (select count(*) from lots where status = 'closed' and winner_bid_id is not null)
  ) into result;
  return result;
end;
$$ language plpgsql;
```

### `supabase/migrations/search.sql`

```sql
-- Full-text search en español
alter table lots add column if not exists fts tsvector
  generated always as (
    to_tsvector('spanish', coalesce(title, '') || ' ' || coalesce(description, ''))
  ) stored;

create index idx_lots_fts on lots using gin(fts);

-- Búsqueda con ranking
create or replace function search_lots(
  p_query text default null,
  p_album int default null,
  p_min_price numeric default null,
  p_max_price numeric default null,
  p_status text default null
)
returns setof lots as $$
begin
  return query
  select *
  from lots
  where (p_query is null or fts @@ plainto_tsquery('spanish', p_query))
    and (p_album is null or album = p_album)
    and (p_min_price is null or starting_price >= p_min_price)
    and (p_max_price is null or starting_price <= p_max_price)
    and (p_status is null or status = p_status)
  order by
    case when p_query is not null
      then ts_rank(fts, plainto_tsquery('spanish', p_query))
      else 0
    end desc,
    sort_order asc;
end;
$$ language plpgsql;
```

### `app/actions/search.ts`

```typescript
'use server'
import { createServerSupabase } from '@/lib/supabase/server'
import type { Lot } from '@/types'

export async function searchLots(
  query: string | null,
  album: number | null,
  minPrice: number | null,
  maxPrice: number | null,
  status: string | null
): Promise<Lot[]> {
  const supabase = await createServerSupabase()

  if (query) {
    const { data, error } = await supabase.rpc('search_lots', {
      p_query: query,
      p_album: album,
      p_min_price: minPrice,
      p_max_price: maxPrice,
      p_status: status,
    })
    return (data ?? []) as Lot[]
  }

  // Sin query textual: filtros simples
  let q = supabase.from('lots').select('*')
  if (album) q = q.eq('album', album)
  if (minPrice) q = q.gte('starting_price', minPrice)
  if (maxPrice) q = q.lte('starting_price', maxPrice)
  if (status) q = q.eq('status', status)

  const { data } = await q.order('sort_order')
  return (data ?? []) as Lot[]
}
```

### `app/lote/catalogo/[album]/route.tsx` (endpoint que devuelve el PDF)

```tsx
import { NextResponse } from 'next/server'
import { renderToBuffer } from '@react-pdf/renderer'
import { CatalogoPDF } from './pdf'
import { createServerSupabase } from '@/lib/supabase/server'

export const dynamic = 'force-dynamic'

export async function GET(
  req: Request,
  { params }: { params: { album: string } }
) {
  const album = Number(params.album)
  if (![1, 2, 3, 4].includes(album)) {
    return NextResponse.json({ error: 'Álbum inválido' }, { status: 400 })
  }

  const supabase = await createServerSupabase()
  const { data: lots } = await supabase
    .from('lots')
    .select('*')
    .eq('album', album)
    .order('sort_order')

  const buffer = await renderToBuffer(
    <CatalogoPDF lots={lots ?? []} album={album} />
  )

  return new NextResponse(buffer, {
    headers: {
      'Content-Type': 'application/pdf',
      'Content-Disposition': `inline; filename="remate-59-album-${album}.pdf"`,
      'Cache-Control': 'public, max-age=300',
    },
  })
}
```
