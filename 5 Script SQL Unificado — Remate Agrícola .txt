-- 5.1 Calcular incremento según precio de partida
create or replace function get_bid_increment(p_starting_price numeric)
returns numeric as $$
begin
  return case
    when p_starting_price < 200 then 10
    when p_starting_price <= 5000 then 50
    else 100
  end;
end;
$$ language plpgsql immutable;

-- 5.2 Procesar puja (ATÓMICA — evita race conditions)
create or replace function place_bid(
  p_lot_id bigint,
  p_amount numeric,
  p_bidder_name text,
  p_session_id text,
  p_phone text default null,
  p_email text default null
)
returns jsonb as $$
declare
  v_lot lots%rowtype;
  v_current_price numeric;
  v_increment numeric;
  v_min_bid numeric;
  v_has_prior_bid boolean;
  v_grace_cutoff timestamptz;
  v_new_bid_id bigint;
  v_extended boolean := false;
begin
  -- Bloquear fila del lote (race-condition-proof)
  select * into v_lot from lots where id = p_lot_id for update;

  if not found then
    return jsonb_build_object('ok', false, 'error', 'Lote no encontrado');
  end if;

  if v_lot.status = 'closed' then
    return jsonb_build_object('ok', false, 'error', 'Este lote ya cerró');
  end if;

  -- Calcular precio actual
  select coalesce(max(amount), v_lot.starting_price)
    into v_current_price from bids where lot_id = p_lot_id;

  v_increment := get_bid_increment(v_lot.starting_price);

  if v_current_price = v_lot.starting_price then
    v_min_bid := v_lot.starting_price;
  else
    v_min_bid := v_current_price + v_increment;
  end if;

  if p_amount < v_min_bid then
    return jsonb_build_object('ok', false, 'error',
      'La puja mínima es US$ ' || v_min_bid);
  end if;

  -- Validar múltiplo del incremento (si ya hay pujas)
  if v_current_price > v_lot.starting_price
     and (p_amount - v_current_price) % v_increment <> 0 then
    return jsonb_build_object('ok', false, 'error',
      'La puja debe ser múltiplo de US$ ' || v_increment);
  end if;

  -- Regla de la "última hora" (servidor como fuente de verdad)
  v_grace_cutoff := v_lot.base_closing_time - interval '1 hour';
  if now() >= v_grace_cutoff then
    select exists(
      select 1 from bids
      where lot_id = p_lot_id
        and bidder_session_id = p_session_id
        and created_at < v_grace_cutoff
    ) into v_has_prior_bid;

    if not v_has_prior_bid then
      return jsonb_build_object('ok', false, 'error',
        'Debe haber ofertado antes de las 19:00 para participar en la última hora de este lote');
    end if;
  end if;

  -- Upsert bidder
  insert into bidders (session_id, name, phone, email)
  values (p_session_id, p_bidder_name, p_phone, p_email)
  on conflict (session_id) do update set
    name = excluded.name,
    phone = coalesce(excluded.phone, bidders.phone),
    email = coalesce(excluded.email, bidders.email),
    updated_at = now();

  -- Insertar puja
  insert into bids (lot_id, bidder_name, bidder_session_id, amount)
  values (p_lot_id, p_bidder_name, p_session_id, p_amount)
  returning id into v_new_bid_id;

  -- Prórroga automática: si quedan < 10 min, extender a 10 min
  if v_lot.closing_time - now() < interval '10 minutes' then
    update lots set
      closing_time = now() + interval '10 minutes',
      status = 'extended',
      updated_at = now()
    where id = p_lot_id;
    v_extended := true;
  else
    update lots set updated_at = now() where id = p_lot_id;
  end if;

  return jsonb_build_object(
    'ok', true,
    'bid_id', v_new_bid_id,
    'new_price', p_amount,
    'extended', v_extended,
    'new_closing_time', (select closing_time from lots where id = p_lot_id)
  );
end;
$$ language plpgsql security definer;

-- 5.3 Cerrar lote y declarar ganador
create or replace function close_lot(p_lot_id bigint)
returns jsonb as $$
declare
  v_winner_bid_id bigint;
  v_winner_name text;
  v_winner_amount numeric;
begin
  select id, bidder_name, amount into v_winner_bid_id, v_winner_name, v_winner_amount
  from bids where lot_id = p_lot_id
  order by amount desc, created_at desc limit 1;

  update lots set
    status = 'closed',
    winner_bid_id = v_winner_bid_id,
    updated_at = now()
  where id = p_lot_id
    and status <> 'closed'
    and closing_time <= now();

  if v_winner_bid_id is not null then
    return jsonb_build_object('ok', true, 'winner', v_winner_name, 'amount', v_winner_amount);
  else
    return jsonb_build_object('ok', true, 'winner', null, 'message', 'Sin pujas');
  end if;
end;
$$ language plpgsql security definer;

create or replace function search_lots(
  p_query text default null,
  p_album int default null,
  p_min_price numeric default null,
  p_max_price numeric default null,
  p_status text default null
)
returns setof lots as $$
begin
  return query
  select * from lots
  where (p_query is null or fts @@ plainto_tsquery('spanish', p_query))
    and (p_album is null or album = p_album)
    and (p_min_price is null or starting_price >= p_min_price)
    and (p_max_price is null or starting_price <= p_max_price)
    and (p_status is null or status = p_status)
  order by
    case when p_query is not null
      then ts_rank(fts, plainto_tsquery('spanish', p_query))
      else 0
    end desc,
    album asc,
    sort_order asc;
end;
$$ language plpgsql;

-- PARTE 7: ANALYTICS — Materialized View

drop materialized view if exists lot_stats;
create materialized view lot_stats as
select
  l.id as lot_id,
  l.title,
  l.album,
  l.starting_price,
  l.status,
  (select count(*) from page_views pv where pv.lot_id = l.id) as total_views,
  (select count(*) from bids b where b.lot_id = l.id) as total_bids,
  (select count(distinct bidder_session_id) from bids where lot_id = l.id) as unique_bidders,
  (select max(amount) from bids where lot_id = l.id) as highest_bid,
  case when exists (select 1 from bids where lot_id = l.id) then 1 else 0 end as converted
from lots l;

create unique index if not exists idx_lot_stats_id on lot_stats(lot_id);

create or replace function refresh_lot_stats() returns void as $$
begin
  refresh materialized view concurrently lot_stats;
end;
$$ language plpgsql;

-- 7.1 Stats globales (últimos 7 días)
create or replace function get_global_stats() returns jsonb as $$
declare result jsonb;
begin
  select jsonb_build_object(
    'total_views', (select count(*) from page_views where created_at > now() - interval '7 days'),
    'unique_visitors', (select count(distinct session_id) from page_views where created_at > now() - interval '7 days'),
    'total_bids', (select count(*) from bids where created_at > now() - interval '7 days'),
    'active_bidders', (select count(distinct bidder_session_id) from bids where created_at > now() - interval '7 days'),
    'revenue_potential', (select coalesce(sum(max_bid), 0) from (select max(amount) as max_bid from bids group by lot_id) s),
    'conversion_rate', (
      select round(100.0 * count(*) filter (where converted = 1) / nullif(count(*), 0), 1)
      from lot_stats
    )
  ) into result;
  return result;
end;
$$ language plpgsql;

-- 7.2 Embudo de conversión
create or replace function get_conversion_funnel() returns jsonb as $$
declare result jsonb;
begin
  select jsonb_build_object(
    'lot_views', (select count(*) from page_views),
    'unique_viewers', (select count(distinct session_id) from page_views),
    'bidders', (select count(distinct bidder_session_id) from bids),
    'total_bids', (select count(*) from bids),
    'won_lots', (select count(*) from lots where status = 'closed' and winner_bid_id is not null)
  ) into result;
  return result;
end;
$$ language plpgsql;

-- PARTE 8: TRIGGER DE NOTIFICACIONES (al ser superado)
-- =============================================================================
create or replace function enqueue_outbid_notification()
returns trigger as $$
declare
  prev_bid record;
begin
  -- Buscar la puja inmediatamente anterior (del nuevo líder)
  select * into prev_bid from bids
    where lot_id = NEW.lot_id and id <> NEW.id
    order by amount desc, created_at desc limit 1;

  if prev_bid is null then return NEW; end if;

  -- Email
  insert into notification_queue (bidder_session_id, lot_id, channel, recipient, subject, body)
  select
    bd.session_id, NEW.lot_id, 'email', bd.email,
    '⚠️ Te superaron en: ' || l.title,
    'Hola ' || bd.name || ', tu puja de US$ ' || prev_bid.amount ||
    ' fue superada por US$ ' || NEW.amount || ' en "' || l.title || '". Pujá de nuevo.'
  from bids b
  join bidders bd on bd.session_id = b.bidder_session_id
  join lots l on l.id = NEW.lot_id
  where b.id = prev_bid.id
    and bd.email is not null
    and bd.notify_on_outbid = true
    and bd.session_id <> NEW.bidder_session_id;

  -- WhatsApp
  insert into notification_queue (bidder_session_id, lot_id, channel, recipient, body)
  select
    bd.session_id, NEW.lot_id, 'whatsapp', bd.whatsapp,
    '🚜 Remate #59 - Te superaron en: ' || l.title ||
    '. Nueva puja: US$ ' || NEW.amount
  from bids b
  join bidders bd on bd.session_id = b.bidder_session_id
  join lots l on l.id = NEW.lot_id
  where b.id = prev_bid.id
    and bd.whatsapp is not null
    and bd.notify_on_outbid = true
    and bd.session_id <> NEW.bidder_session_id;

  return NEW;
end;
$$ language plpgsql;

drop trigger if exists trg_enqueue_outbid on bids;
create trigger trg_enqueue_outbid
  after insert on bids
  for each row execute function enqueue_outbid_notification();

## 🔄 Re-ejecuciones seguras

El script es **idempotente**:
- Usa `CREATE OR REPLACE` para funciones
- Usa `IF NOT EXISTS` para tablas, columnas e índices
- El bloque de seed detecta si ya hay datos y los salta
- Los cron jobs envuelven en `EXCEPTION` por si ya existen

Podés ejecutarlo mil veces sin romper nada. Si querés **resetear todo** y empezar de cero:

```sql
-- ⚠️ PELIGROSO: borra todo
drop table if exists bids, notification_queue, payments, page_views, bidders, lots cascade;
drop materialized view if exists lot_stats;
-- Y luego volvé a correr setup.sql
```
