## 🚀 2. Mejoras críticas sobre la propuesta de DeepSeek

| Problema del original | Mi solución |
|---|---|
| ⚠️ Validación de pujas en API REST vulnerable a race conditions | ✅ **RPC de PostgreSQL con transacción + row locking** |
| ⚠️ Confía en el reloj del cliente para la "última hora" | ✅ **Todo se valida con `NOW()` del servidor** |
| ⚠️ `localStorage` con ID aleatorio = fácil de falsificar | ✅ **Fingerprint + cookie firmada + rate limiting** |
| ⚠️ Tailwind puro = mucho código repetido | ✅ **shadcn/ui + Framer Motion** |
| ⚠️ Sin feedback sonoro/visual al ser superado | ✅ **Toasts + sonido + vibración mobile** |
| ⚠️ Admin con login hardcodeado en cliente | ✅ **Middleware Next.js + JWT firmado en servidor** |
| ⚠️ Sin manejo de "casi ganar" | ✅ **Modal de "¡GANASTE!" con confetti + datos de contacto** |
| ⚠️ Sin catálogo imprimible | ✅ **Generación de PDF del álbum** |

```sql
-- =========================================================
-- REMATE AGRÍCOLA VIRTUAL - Schema + RPC de pujas atómico
-- =========================================================

-- Extensión necesaria
create extension if not exists pgcrypto;

-- Tabla de lotes
create table lots (
  id bigserial primary key,
  title text not null,
  description text,
  starting_price numeric(12,2) not null check (starting_price >= 0),
  album smallint not null check (album between 1 and 4),
  image_urls text[] not null default '{}',
  base_closing_time timestamptz not null,
  closing_time timestamptz not null,
  status text not null default 'active'
    check (status in ('active','extended','closed')),
  winner_bid_id bigint,
  sort_order int default 0,
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

create index idx_lots_album on lots(album);
create index idx_lots_closing on lots(closing_time) where status <> 'closed';

-- Tabla de pujas
create table bids (
  id bigserial primary key,
  lot_id bigint not null references lots(id) on delete cascade,
  bidder_name text not null,
  bidder_session_id text not null,
  amount numeric(12,2) not null,
  created_at timestamptz default now()
);

create index idx_bids_lot on bids(lot_id, created_at desc);
create index idx_bids_session on bids(lot_id, bidder_session_id, created_at);

-- Tabla de sesiones (para identidad del pujador)
create table bidders (
  session_id text primary key,
  name text not null,
  phone text,
  whatsapp text,
  created_at timestamptz default now()
);

-- =========================================================
-- FUNCIÓN: obtener incremento según precio de partida
-- =========================================================
create or replace function get_bid_increment(starting_price numeric)
returns numeric as $$
begin
  return case
    when starting_price < 200 then 10
    when starting_price <= 5000 then 50
    else 100
  end;
end;
$$ language plpgsql immutable;

-- =========================================================
-- FUNCIÓN PRINCIPAL: procesar puja (ATÓMICA)
-- Evita race conditions con SELECT FOR UPDATE
-- =========================================================
create or replace function place_bid(
  p_lot_id bigint,
  p_amount numeric,
  p_bidder_name text,
  p_session_id text,
  p_phone text default null
)
returns jsonb as $$
declare
  v_lot lots%rowtype;
  v_current_price numeric;
  v_increment numeric;
  v_min_bid numeric;
  v_has_prior_bid boolean;
  v_grace_cutoff timestamptz;
  v_new_bid_id bigint;
  v_extended boolean := false;
begin
  -- 1. Bloquear fila del lote (evita race conditions)
  select * into v_lot from lots where id = p_lot_id for update;

  if not found then
    return jsonb_build_object('ok', false, 'error', 'Lote no encontrado');
  end if;

  if v_lot.status = 'closed' then
    return jsonb_build_object('ok', false, 'error', 'Este lote ya cerró');
  end if;

  -- 2. Calcular precio actual
  select coalesce(max(amount), v_lot.starting_price)
    into v_current_price
    from bids where lot_id = p_lot_id;

  v_increment := get_bid_increment(v_lot.starting_price);

  if v_current_price = v_lot.starting_price then
    -- Primera puja: debe ser >= precio de partida y múltiplo del incremento
    v_min_bid := v_lot.starting_price;
    if p_amount < v_min_bid then
      return jsonb_build_object('ok', false, 'error',
        'La puja mínima es US$ ' || v_min_bid);
    end if;
  else
    v_min_bid := v_current_price + v_increment;
    if p_amount < v_min_bid then
      return jsonb_build_object('ok', false, 'error',
        'La puja mínima es US$ ' || v_min_bid ||
        ' (actual: US$ ' || v_current_price ||
        ' + incremento de US$ ' || v_increment || ')');
    end if;
    -- Verificar que sea múltiplo del incremento sobre el precio actual
    if (p_amount - v_current_price) % v_increment <> 0 then
      return jsonb_build_object('ok', false, 'error',
        'La puja debe ser múltiplo de US$ ' || v_increment);
    end if;
  end if;

  -- 3. Regla de la "última hora" (HORA DEL SERVIDOR, no del cliente)
  v_grace_cutoff := v_lot.base_closing_time - interval '1 hour';
  if now() >= v_grace_cutoff then
    select exists(
      select 1 from bids
      where lot_id = p_lot_id
        and bidder_session_id = p_session_id
        and created_at < v_grace_cutoff
    ) into v_has_prior_bid;

    if not v_has_prior_bid then
      return jsonb_build_object('ok', false, 'error',
        'Debe haber ofertado antes de las 19:00 para participar en la última hora de este lote');
    end if;
  end if;

  -- 4. Upsert del bidder
  insert into bidders (session_id, name, phone)
  values (p_session_id, p_bidder_name, p_phone)
  on conflict (session_id) do update
    set name = excluded.name,
        phone = coalesce(excluded.phone, bidders.phone);

  -- 5. Insertar la puja
  insert into bids (lot_id, bidder_name, bidder_session_id, amount)
  values (p_lot_id, p_bidder_name, p_session_id, p_amount)
  returning id into v_new_bid_id;

  -- 6. Prórroga automática: si faltan <10 min, extender a 10 min
  if v_lot.closing_time - now() < interval '10 minutes' then
    update lots set
      closing_time = now() + interval '10 minutes',
      status = 'extended',
      updated_at = now()
    where id = p_lot_id;
    v_extended := true;
  else
    update lots set updated_at = now() where id = p_lot_id;
  end if;

  return jsonb_build_object(
    'ok', true,
    'bid_id', v_new_bid_id,
    'new_price', p_amount,
    'extended', v_extended,
    'new_closing_time',
      (select closing_time from lots where id = p_lot_id)
  );
end;
$$ language plpgsql security definer;

-- =========================================================
-- FUNCIÓN: cerrar lote y declarar ganador
-- =========================================================
create or replace function close_lot(p_lot_id bigint)
returns jsonb as $$
declare
  v_winner_bid_id bigint;
  v_winner_name text;
  v_winner_amount numeric;
begin
  select id, bidder_name, amount into v_winner_bid_id, v_winner_name, v_winner_amount
  from bids where lot_id = p_lot_id
  order by amount desc, created_at desc limit 1;

  update lots set
    status = 'closed',
    winner_bid_id = v_winner_bid_id,
    updated_at = now()
  where id = p_lot_id and status <> 'closed'
    and closing_time <= now();

  if v_winner_bid_id is not null then
    return jsonb_build_object(
      'ok', true, 'winner', v_winner_name, 'amount', v_winner_amount
    );
  else
    return jsonb_build_object('ok', true, 'winner', null, 'message', 'Sin pujas');
  end if;
end;
$$ language plpgsql security definer;

-- =========================================================
-- REALTIME: habilitar para lots y bids
-- =========================================================
alter publication supabase_realtime add table lots;
alter publication supabase_realtime add table bids;

-- RLS (permisivo para MVP - luego ajustar)
alter table lots enable row level security;
alter table bids enable row level security;

create policy "Lectura pública de lotes" on lots for select using (true);
create policy "Lectura pública de pujas" on bids for select using (true);
```

### `lib/auction/rules.ts` — Motor de reglas de negocio

```typescript
// Motor de reglas del remate - SINGLE SOURCE OF TRUTH
export const AUCTION_RULES = {
  // Incrementos según precio de partida
  getIncrement(startingPrice: number): number {
    if (startingPrice < 200) return 10
    if (startingPrice <= 5000) return 50
    return 100
  },

  // Calcular próxima puja válida
  getNextBid(currentPrice: number, startingPrice: number): number {
    if (currentPrice < startingPrice) return startingPrice
    return currentPrice + this.getIncrement(startingPrice)
  },

  // Generar opciones de puja rápida (+1, +3, +5 incrementos)
  getQuickBids(currentPrice: number, startingPrice: number): number[] {
    const inc = this.getIncrement(startingPrice)
    const base = Math.max(currentPrice, startingPrice)
    return [1, 3, 5].map(m => base + inc * m)
  },

  // Comisión según método de pago
  getCommission(amount: number, method: 'cash' | 'mercadopago') {
    const rate = method === 'cash' ? 0.12 : 0.19
    const commission = amount * rate
    return { rate, commission, total: amount + commission }
  },

  // Umbrales del countdown
  getCountdownState(secondsLeft: number): 'safe' | 'warning' | 'critical' {
    if (secondsLeft > 600) return 'safe'       // > 10 min
    if (secondsLeft > 60) return 'warning'     // 1-10 min
    return 'critical'                          // < 1 min
  },

  // Verificar si está en "última hora"
  isInGracePeriod(now: Date, baseClosingTime: Date): boolean {
    const graceStart = new Date(baseClosingTime.getTime() - 60 * 60 * 1000)
    return now >= graceStart
  },
}

export const formatCurrency = (n: number) =>
  new Intl.NumberFormat('es-UY', {
    style: 'currency', currency: 'USD',
    minimumFractionDigits: 0, maximumFractionDigits: 0,
  }).format(n)
```

### `components/auction/CountdownTimer.tsx` — La estrella del UX

```tsx
// components/auction/CountdownTimer.tsx
'use client'
import { useEffect, useState } from 'react'
import { AUCTION_RULES } from '@/lib/auction/rules'

interface Props {
  closingTime: string
  baseClosingTime: string
  status: 'active' | 'extended' | 'closed'
  onExpire?: () => void
}

export function CountdownTimer({ closingTime, baseClosingTime, status, onExpire }: Props) {
  const [now, setNow] = useState(Date.now())

  useEffect(() => {
    const id = setInterval(() => setNow(Date.now()), 1000)
    return () => clearInterval(id)
  }, [])

  const target = new Date(closingTime).getTime()
  const diff = Math.max(0, Math.floor((target - now) / 1000))

  useEffect(() => {
    if (diff === 0 && onExpire) onExpire()
  }, [diff, onExpire])

  if (status === 'closed' || diff === 0) {
    return (
      <div className="bg-earth text-cream text-center py-3 px-4 rounded-xl font-bold">
        🏁 LOTE CERRADO
      </div>
    )
  }

  const days = Math.floor(diff / 86400)
  const hours = Math.floor((diff % 86400) / 3600)
  const mins = Math.floor((diff % 3600) / 60)
  const secs = diff % 60
  const state = AUCTION_RULES.getCountdownState(diff)
  const isExtended = status === 'extended'

  const colors = {
    safe: 'bg-forest text-white',
    warning: 'bg-wheat text-earth pulse-warning',
    critical: 'bg-alert text-white pulse-critical',
  }

  return (
    <div className={`rounded-2xl p-4 ${colors[state]} shadow-lg`}>
      {isExtended && (
        <div className="text-center text-xs font-bold mb-2 tracking-widest">
          ⚡ PRÓRROGA ACTIVA — 10 MIN EXTRA
        </div>
      )}
      <div className="text-center text-[10px] uppercase tracking-widest opacity-80 mb-1">
        {state === 'critical' ? '⏰ Último minuto' : 'Tiempo restante'}
      </div>
      <div className="flex items-center justify-center gap-2 font-display font-bold">
        {days > 0 && (
          <TimeBlock value={days} label="días" />
        )}
        <TimeBlock value={hours} label="hrs" />
        <span className="text-2xl opacity-50">:</span>
        <TimeBlock value={mins} label="min" />
        <span className="text-2xl opacity-50">:</span>
        <TimeBlock value={secs} label="seg" highlight={state === 'critical'} />
      </div>
    </div>
  )
}

function TimeBlock({ value, label, highlight }: {

}) {
  return (
    <div className="flex flex-col items-center">
```

### `app/actions/bids.ts` — Server Action (más seguro que API route)

```typescript
// app/actions/bids.ts
'use server'
import { cookies, headers } from 'next/headers'
import { createServerSupabase } from '@/lib/supabase/server'
import { z } from 'zod'
import { revalidatePath } from 'next/cache'

const schema = z.object({
  lotId: z.number().int().positive(),
  amount: z.number().positive(),
  bidderName: z.string().min(2).max(60).trim(),
  phone: z.string().optional(),
})

export async function placeBid(
  lotId: number, amount: number, bidderName: string, phone?: string
) {
  const validated = schema.safeParse({ lotId, amount, bidderName, phone })
  if (!validated.success) {
    return { ok: false, error: 'Datos inválidos' }
  }

  // Rate limiting simple: 1 puja cada 2 seg por sesión
  const cookieStore = await cookies()
  const lastBid = cookieStore.get(`last_bid_${lotId}`)
  if (lastBid && Date.now() - Number(lastBid.value) < 2000) {
    return { ok: false, error: 'Esperá unos segundos antes de volver a pujar' }
  }

  let sessionId = cookieStore.get('bidder_sid')?.value
  if (!sessionId) {
    sessionId = crypto.randomUUID()
    ;(await cookies()).set('bidder_sid', sessionId, {
      httpOnly: true, secure: true, sameSite: 'lax', maxAge: 60 * 60 * 24 * 30,
    })
  }

  const supabase = await createServerSupabase()
  const { data, error } = await supabase.rpc('place_bid', {
    p_lot_id: lotId,
    p_amount: amount,
    p_bidder_name: bidderName,
    p_session_id: sessionId,
    p_phone: phone ?? null,
  })

  if (error || !data?.ok) {
    return { ok: false, error: data?.error || error?.message || 'Error' }
  }

  ;(await cookies()).set(`last_bid_${lotId}`, String(Date.now()), { maxAge: 10 })
  revalidatePath(`/lote/${lotId}`)
  revalidatePath('/')
  return { ok: true, ...data }
}
```

### `components/auction/RealtimeLotSubscriber.tsx`

```tsx
'use client'
import { useEffect } from 'react'
import { createClient } from '@/lib/supabase/client'
import { useToast } from '@/components/ui/use-toast'
import { useRouter } from 'next/navigation'
import { playBidSound } from '@/lib/sounds'

export function RealtimeLotSubscriber({ lotId }: { lotId: number }) {
  const { toast } = useToast()
  const router = useRouter()

  useEffect(() => {
    const supabase = createClient()

    const lotChannel = supabase
      .channel(`lot-${lotId}`)
      .on('postgres_changes',
        { event: 'UPDATE', schema: 'public', table: 'lots', filter: `id=eq.${lotId}` },
        (payload: any) => {
          if (payload.new?.status === 'extended' && payload.old?.status === 'active') {
            toast({ title: '⚡ ¡PRÓRROGA!', description: 'El lote se extendió 10 minutos' })
          }
          if (payload.new?.status === 'closed') {
            router.refresh()
          }
        })
      .subscribe()

    const bidsChannel = supabase
      .channel(`bids-${lotId}`)
      .on('postgres_changes',
        { event: 'INSERT', schema: 'public', table: 'bids', filter: `lot_id=eq.${lotId}` },
        () => {
          playBidSound()
          router.refresh()
        })
      .subscribe()

    return () => {
      supabase.removeChannel(lotChannel)
      supabase.removeChannel(bidsChannel)
    }
  }, [lotId, toast, router])

  return null
}
```

```typescript
let ctx: AudioContext | null = null
const getCtx = () => {
  if (!ctx) ctx = new (window.AudioContext || (window as any).webkitAudioContext)()
  return ctx
}

export function playBidSound() {
  try {
    const c = getCtx()
    const o = c.createOscillator()
    const g = c.createGain()
    o.connect(g); g.connect(c.destination)
    o.frequency.setValueAtTime(660, c.currentTime)
    o.frequency.exponentialRampToValueAtTime(880, c.currentTime + 0.1)
    g.gain.setValueAtTime(0.1, c.currentTime)
    g.gain.exponentialRampToValueAtTime(0.001, c.currentTime + 0.2)
    o.start(); o.stop(c.currentTime + 0.2)
  } catch {}
}

export function playWinSound() {
  try {
    const c = getCtx()
    ;[523, 659, 784, 1047].forEach((freq, i) => {
      const o = c.createOscillator()
      const g = c.createGain()
      o.connect(g); g.connect(c.destination)
      o.frequency.value = freq
      g.gain.setValueAtTime(0.15, c.currentTime + i * 0.12)
      g.gain.exponentialRampToValueAtTime(0.001, c.currentTime + i * 0.12 + 0.3)
      o.start(c.currentTime + i * 0.12)
      o.stop(c.currentTime + i * 0.12 + 0.3)
    })
  } catch {}
```

```tsx
'use client'
import { createContext, useContext, useEffect, useState } from 'react'

interface Bidder { name: string; phone: string }
interface Ctx {
  bidder: Bidder
  setBidderName: (n: string) => void
  setBidderPhone: (p: string) => void
}

const BidderCtx = createContext<Ctx | null>(null)

export function BidderProvider({ children }: { children: React.ReactNode }) {
  const [bidder, setBidder] = useState<Bidder>({ name: '', phone: '' })

  useEffect(() => {
    const stored = localStorage.getItem('bidder')
    if (stored) setBidder(JSON.parse(stored))
  }, [])

  const update = (patch: Partial<Bidder>) => {
    setBidder(prev => {
      const next = { ...prev, ...patch }
      localStorage.setItem('bidder', JSON.stringify(next))
      return next
    })
  }

  return (
    <BidderCtx.Provider value={{
      bidder,
      setBidderName: n => update({ name: n }),
      setBidderPhone: p => update({ phone: p }),
    }}>
      {children}
    </BidderCtx.Provider>
  )
}

export const useBidder = () => {
  const ctx = useContext(BidderCtx)
  if (!ctx) throw new Error('useBidder must be used within BidderProvider')
  return ctx
}
```

```typescript
'use server'
import { cookies } from 'next/headers'
import { redirect } from 'next/navigation'
import crypto from 'crypto'

export async function adminLogin(formData: FormData) {
  const u = formData.get('user') as string
  const p = formData.get('pass') as string
  if (u === process.env.ADMIN_USER && p === process.env.ADMIN_PASS) {
    const token = crypto
      .createHmac('sha256', process.env.ADMIN_SECRET || 'dev-secret')
      .update(u + p).digest('hex')

    redirect('/admin/dashboard')
```
