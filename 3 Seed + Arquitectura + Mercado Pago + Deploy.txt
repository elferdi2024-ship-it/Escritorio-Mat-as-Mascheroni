````markdown
## Decisiones técnicas clave

| Decisión | Por qué |
|---|---|
| **RPC PostgreSQL para pujas** | Atomicidad real con `SELECT FOR UPDATE`. Cero race conditions aunque 50 personas puje al mismo tiempo en el último segundo. |
| **Supabase Realtime (postgres_changes)** | Sin infra adicional. Sincroniza precio, historial y cronómetro en <200ms. |
| **Next.js Server Actions (no API routes)** | Menos código, validación compartida, `revalidatePath` automático. |
| **Cookie firmada para sesión de bidder** | Más segura que `localStorage` (el cliente no puede falsificar el session_id). |
| **Todo validado en servidor con `NOW()`** | El reloj del cliente nunca es fuente de verdad. Evita que alguien cambie su hora local. |
| **ISR con revalidate=60 + Realtime** | SEO perfecto + updates en vivo. Lo mejor de ambos mundos. |
| **Tailwind + shadcn/ui patterns** | Consistencia visual + bundle liviano. |

## Escalabilidad

| Hasta | Solución actual |
|---|---|
| 500 usuarios concurrentes | Supabase free tier + Vercel Hobby ✅ |
| 5.000 concurrentes | Supabase Pro (US$ 25/mes) + Vercel Pro ✅ |
| 50.000 concurrentes | Considerar migrar Realtime a Pusher/Ably + Redis cache |
| 1M pujas/día | Particionar tabla `bids` por `lot_id` y archivar remates pasados |

## Seguridad

- ✅ Admin protegido por cookie HMAC firmada
- ✅ Rate limiting por sesión (1 puja cada 2s por lote)
- ✅ Validación Zod en Server Action + RPC en DB (doble barrera)
- ✅ Row Level Security habilitado (ajustar políticas antes de prod)
- ⚠️ Pendiente: CAPTCHA opcional si hay ataques de bots
````

### `app/actions/checkout.ts`

```typescript
'use server'
import { cookies } from 'next/headers'
import { createServerSupabase } from '@/lib/supabase/server'
import { preferenceClient, MP_COMMISSION_RATE } from '@/lib/mercadopago/client'
import { redirect } from 'next/navigation'

export async function createMercadoPagoCheckout(lotId: number) {
  const supabase = await createServerSupabase()

  // Obtener lote + puja ganadora
  const { data: lot } = await supabase
    .from('lots').select('*, winner_bid:bids!winner_bid_id(*)')
    .eq('id', lotId).single()

  if (!lot || lot.status !== 'closed') {
    return { ok: false, error: 'Lote no disponible' }
  }

  const winnerBid = lot.winner_bid as any
  const cookieSessionId = (await cookies()).get('bidder_sid')?.value

  // Solo el ganador puede pagar
  if (!winnerBid || winnerBid.bidder_session_id !== cookieSessionId) {
    return { ok: false, error: 'Solo el ganador puede pagar este lote' }
  }

  const commission = winnerBid.amount * MP_COMMISSION_RATE
  const total = winnerBid.amount + commission

  // Crear preferencia de pago
  const preference = await preferenceClient.create({
    body: {
      items: [{
        id: `remate-59-lote-${lot.id}`,
        title: `Remate #59 · Lote ${lot.id}: ${lot.title}`,
        description: `Adjudicado por US$ ${winnerBid.amount} + 19% comisión MP`,
        quantity: 1,
        currency_id: 'USD',
        unit_price: Number(total.toFixed(2)),
      }],
      payer: {
        name: winnerBid.bidder_name,
        email: `bidder-${winnerBid.bidder_session_id.slice(0,8)}@remate.local`, // Reemplazar por email real cuando lo tengamos
      },
      external_reference: `lot-${lot.id}-bid-${winnerBid.id}`,
      back_urls: {
        success: `${process.env.NEXT_PUBLIC_SITE_URL}/pago/exitoso?lot=${lot.id}`,
        failure: `${process.env.NEXT_PUBLIC_SITE_URL}/pago/fallido?lot=${lot.id}`,
        pending: `${process.env.NEXT_PUBLIC_SITE_URL}/pago/pendiente?lot=${lot.id}`,
      },
      auto_return: 'approved',
      notification_url: `${process.env.NEXT_PUBLIC_SITE_URL}/api/webhooks/mercadopago`,
    },
  })

  return { ok: true, initPoint: preference.init_point }
}

export async function redirectToCheckout(lotId: number) {
  const res = await createMercadoPagoCheckout(lotId)
  if (!res.ok || !res.initPoint) {
    throw new Error(res.error || 'Error creando preferencia')
  }
  redirect(res.initPoint)
}
```

### `app/api/webhooks/mercadopago/route.ts`

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { paymentClient } from '@/lib/mercadopago/client'
import { createServerSupabase } from '@/lib/supabase/server'

export async function POST(req: NextRequest) {
  try {
    const body = await req.json()
    const topic = body.topic || body.type
    const id = body.data?.id || body.data?.id

    if (topic === 'payment' && id) {
      const payment = await paymentClient.get({ id })

      if (payment.status === 'approved') {
        const [lotIdStr] = (payment.external_reference || '').split('-').slice(1)
        const lotId = Number(lotIdStr)

        if (lotId) {
          const supabase = await createServerSupabase()
          await supabase.from('payments').insert({
            lot_id: lotId,
            mp_payment_id: id,
            amount: Number(payment.transaction_amount),
            status: 'approved',
            paid_at: new Date().toISOString(),
          })

          // TODO: Enviar email/WhatsApp de confirmación al ganador
        }
      }
    }

    return NextResponse.json({ ok: true })
  } catch (error) {
    console.error('Webhook MP error:', error)
    return NextResponse.json({ error: 'error' }, { status: 500 })
  }
}
```
