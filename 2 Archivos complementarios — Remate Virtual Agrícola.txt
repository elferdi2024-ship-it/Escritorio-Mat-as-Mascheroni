```tsx
import { cva, type VariantProps } from 'class-variance-authority'

const toastVariants = cva(
  'group pointer-events-auto relative flex w-full items-center justify-between space-x-4 overflow-hidden rounded-xl border p-4 pr-8 shadow-lg transition-all',
  {
    variants: {
      variant: {
        default: 'border-cream-dark bg-white text-earth',

      },
    },
    defaultVariants: { variant: 'default' },
  }

  extends React.HTMLAttributes<HTMLDivElement>,
    VariantProps<typeof toastVariants> {
  onClose?: () => void
}

    <div ref={ref} className={cn(toastVariants({ variant }), className)} {...props}>
      <div className="flex-1 text-sm">{children}</div>

        <button
          onClick={onClose}
          className="absolute right-2 top-2 rounded-md p-1 opacity-50 hover:opacity-100"
        >
          <X className="h-4 w-4" />
        </button>
      )}
    </div>
  )
```

### `components/ui/use-toast.ts`

```typescript
'use client'
import { useEffect, useState } from 'react'

export type ToastVariant = 'default' | 'destructive' | 'success'
export interface ToastMessage {
  id: string
  title?: string
  description?: string
  variant?: ToastVariant
  duration?: number
}

type Listener = (toasts: ToastMessage[]) => void
let toasts: ToastMessage[] = []
const listeners = new Set<Listener>()

function emit() { listeners.forEach(l => l([...toasts])) }

export function useToast() {
  const [state, setState] = useState<ToastMessage[]>([])

  useEffect(() => {
    const listener: Listener = setState
    listeners.add(listener)
    setState([...toasts])
    return () => { listeners.delete(listener) }
  }, [])

  return {
    toasts: state,
    toast: (msg: Omit<ToastMessage, 'id'>) => {
      const id = Math.random().toString(36).slice(2)
      const duration = msg.duration ?? 4500
      toasts = [...toasts, { ...msg, id }]
      emit()
      setTimeout(() => {
        toasts = toasts.filter(t => t.id !== id)
        emit()
      }, duration)
      return { id }
    },
    dismiss: (id: string) => {
      toasts = toasts.filter(t => t.id !== id)
      emit()
    },
  }
}
```

```tsx
import { useEffect, useState } from 'react'
import type { Lot } from '@/types'
import { formatCurrency } from '@/lib/auction/rules'

export function LiveTicker({ lots }: { lots: Lot[] }) {
  const [index, setIndex] = useState(0)

  useEffect(() => {
    if (lots.length < 2) return
    const id = setInterval(() => setIndex(i => (i + 1) % lots.length), 3500)
```

```tsx
import { useState } from 'react'
import Image from 'next/image'
import { ChevronLeft, ChevronRight, X, ZoomIn } from 'lucide-react'

interface Props {
  images: string[]
  title: string

export function ImageGallery({ images, title }: Props) {
```

```tsx
export function CountdownBadge({ closingTime, status }: Props) {
  const [now, setNow] = useState(Date.now())
  useEffect(() => {
    const id = setInterval(() => setNow(Date.now()), 1000)
    return () => clearInterval(id)
  }, [])

  if (status === 'closed') {
    return (
      <span className="px-2 py-1 bg-earth text-cream text-[10px] font-bold rounded-md">
        CERRADO
      </span>
    )
  }

  const diff = Math.max(0, Math.floor((new Date(closingTime).getTime() - now) / 1000))
  const state = AUCTION_RULES.getCountdownState(diff)

  const d = Math.floor(diff / 86400)
  const h = Math.floor((diff % 86400) / 3600)
  const m = Math.floor((diff % 3600) / 60)
  const s = diff % 60

  const colors = {
    safe: 'bg-forest text-white',
    warning: 'bg-wheat text-earth',
    critical: 'bg-alert text-white pulse-critical',
  }

  const text = d > 0
    ? `${d}d ${h}h`
    : h > 0
      ? `${h}h ${m}m`
      : `${m}:${String(s).padStart(2, '0')}`

  return (
    <span className={`px-2 py-1 text-[10px] font-bold rounded-md tabular-nums ${colors[state]}`}>
      ⏱ {text}
    </span>
  )
}
```

### `app/actions/admin-lots.ts` — Server Actions CRUD

```typescript
'use server'
import { revalidatePath } from 'next/cache'
import { redirect } from 'next/navigation'
import { cookies } from 'next/headers'
import { createServerSupabase } from '@/lib/supabase/server'
import { z } from 'zod'

const ensureAdmin = async () => {
  const c = await cookies()
  if (c.get('admin_auth')?.value !== process.env.ADMIN_TOKEN) {
    redirect('/admin/login')
  }
}

const lotSchema = z.object({
  title: z.string().min(3).max(200),
  description: z.string().optional(),
  starting_price: z.number().positive(),
  album: z.number().int().min(1).max(4),
  image_urls: z.array(z.string().url()).min(1),
  base_closing_time: z.string().datetime(),
  sort_order: z.number().int().default(0),
})

export async function createLot(formData: FormData) {
  await ensureAdmin()
  const raw = {
    title: formData.get('title') as string,
    description: (formData.get('description') as string) || undefined,
    starting_price: Number(formData.get('starting_price')),
    album: Number(formData.get('album')),
    image_urls: (formData.get('image_urls') as string).split('\n').map(s => s.trim()).filter(Boolean),
    base_closing_time: formData.get('base_closing_time') as string,
    sort_order: Number(formData.get('sort_order') || 0),
  }

  const parsed = lotSchema.safeParse(raw)
  if (!parsed.success) {
    return { ok: false, error: parsed.error.issues[0].message }
  }

  const supabase = await createServerSupabase()
  const { data, error } = await supabase.from('lots').insert({
    ...parsed.data,
    closing_time: parsed.data.base_closing_time,
    status: 'active',
  }).select('id').single()

  if (error) return { ok: false, error: error.message }
  revalidatePath('/admin/lotes')
  revalidatePath('/')
  redirect(`/admin/lotes/${data.id}/editar`)
}

export async function updateLot(id: number, formData: FormData) {
  await ensureAdmin()
  const raw = {
    title: formData.get('title') as string,
    description: (formData.get('description') as string) || undefined,
    starting_price: Number(formData.get('starting_price')),
    album: Number(formData.get('album')),
    image_urls: (formData.get('image_urls') as string).split('\n').map(s => s.trim()).filter(Boolean),
    base_closing_time: formData.get('base_closing_time') as string,
    sort_order: Number(formData.get('sort_order') || 0),
  }

  const parsed = lotSchema.safeParse(raw)
  if (!parsed.success) return { ok: false, error: parsed.error.issues[0].message }

  const supabase = await createServerSupabase()
  const { error } = await supabase.from('lots').update({
    ...parsed.data,
    updated_at: new Date().toISOString(),
  }).eq('id', id)

  if (error) return { ok: false, error: error.message }
  revalidatePath('/admin/lotes')
  revalidatePath(`/lote/${id}`)
  revalidatePath('/')
  return { ok: true }
}

export async function resetLot(id: number) {
  await ensureAdmin()
  const supabase = await createServerSupabase()

  // Obtener base_closing_time para restaurarlo
  const { data: lot } = await supabase.from('lots').select('base_closing_time').eq('id', id).single()
  if (!lot) return

  await supabase.from('bids').delete().eq('lot_id', id)
  await supabase.from('lots').update({
    closing_time: lot.base_closing_time,
    status: 'active',
    winner_bid_id: null,
    updated_at: new Date().toISOString(),
  }).eq('id', id)

  revalidatePath('/admin/lotes')
  revalidatePath(`/lote/${id}`)
  revalidatePath('/')
}

export async function deleteLot(id: number) {
  await ensureAdmin()
  const supabase = await createServerSupabase()
  await supabase.from('bids').delete().eq('lot_id', id)
```

```tsx
function Field({
  label, name, type = 'text', textarea, required, defaultValue, placeholder, step,

function getNextTuesday20() {
  const now = new Date()
  const day = now.getDay()
  const daysUntilTuesday = (2 - day + 7) % 7 || 7
  const next = new Date(now)
  next.setDate(now.getDate() + daysUntilTuesday)
  next.setHours(20, 0, 0, 0)
  return next.toISOString().slice(0, 16)
}
```

### `__tests__/bid-rpc.test.ts` (lógica del RPC simulada en JS)

```typescript
import { describe, it, expect } from 'vitest'
import { AUCTION_RULES } from '../lib/auction/rules'

// Simulador en memoria de la función place_bid para testear reglas sin DB
function simulatePlaceBid(opts: {
  startingPrice: number
  currentPrice: number
  bidAmount: number
  lotStatus: 'active' | 'extended' | 'closed'
  closingTime: Date
  baseClosingTime: Date
  now: Date
  bidderHasPriorBid: boolean
}) {
  if (opts.lotStatus === 'closed') {
    return { ok: false, error: 'Este lote ya cerró' }
  }

  const increment = AUCTION_RULES.getIncrement(opts.startingPrice)
  const minBid = opts.currentPrice < opts.startingPrice
    ? opts.startingPrice
    : opts.currentPrice + increment

  if (opts.bidAmount < minBid) {
    return { ok: false, error: `La puja mínima es US$ ${minBid}` }
  }

  const graceCutoff = new Date(opts.baseClosingTime.getTime() - 60 * 60 * 1000)
  if (opts.now >= graceCutoff && !opts.bidderHasPriorBid) {
    return { ok: false, error: 'Debe haber ofertado antes de las 19:00' }
  }

  const tenMinBeforeClose = new Date(opts.closingTime.getTime() - 10 * 60 * 1000)
  const extended = opts.now >= tenMinBeforeClose
  const newClosing = extended
    ? new Date(opts.now.getTime() + 10 * 60 * 1000)
    : opts.closingTime

  return { ok: true, newPrice: opts.bidAmount, extended, newClosing }
}

describe('Simulación RPC place_bid', () => {
  const baseClosing = new Date('2026-05-19T20:00:00Z')

  it('rechaza si el lote ya cerró', () => {
    const r = simulatePlaceBid({
      startingPrice: 100, currentPrice: 100, bidAmount: 100,
      lotStatus: 'closed', closingTime: baseClosing, baseClosingTime: baseClosing,
      now: new Date('2026-05-19T20:05:00Z'), bidderHasPriorBid: false,
    })
    expect(r.ok).toBe(false)
  })

  it('acepta primera puja igual al precio de partida', () => {
    const r = simulatePlaceBid({
      startingPrice: 100, currentPrice: 0, bidAmount: 100,
      lotStatus: 'active', closingTime: baseClosing, baseClosingTime: baseClosing,
      now: new Date('2026-05-19T17:00:00Z'), bidderHasPriorBid: false,
    })
    expect(r.ok).toBe(true)
  })

  it('rechaza puja menor al incremento', () => {
    const r = simulatePlaceBid({
      startingPrice: 100, currentPrice: 100, bidAmount: 105,
      lotStatus: 'active', closingTime: baseClosing, baseClosingTime: baseClosing,
      now: new Date('2026-05-19T17:00:00Z'), bidderHasPriorBid: false,
    })
    expect(r.ok).toBe(false)
  })

  it('rechaza si está en última hora sin puja previa', () => {
    const r = simulatePlaceBid({
      startingPrice: 100, currentPrice: 100, bidAmount: 110,
      lotStatus: 'active', closingTime: baseClosing, baseClosingTime: baseClosing,
      now: new Date('2026-05-19T19:30:00Z'), bidderHasPriorBid: false,
    })
    expect(r.ok).toBe(false)
  })

  it('acepta si está en última hora CON puja previa', () => {
    const r = simulatePlaceBid({
      startingPrice: 100, currentPrice: 100, bidAmount: 110,
      lotStatus: 'active', closingTime: baseClosing, baseClosingTime: baseClosing,
      now: new Date('2026-05-19T19:30:00Z'), bidderHasPriorBid: true,
    })
    expect(r.ok).toBe(true)
  })

  it('extiende 10 min si la puja llega en los últimos 10 min', () => {
    const r = simulatePlaceBid({
      startingPrice: 100, currentPrice: 100, bidAmount: 110,
      lotStatus: 'active', closingTime: baseClosing, baseClosingTime: baseClosing,

    expect(r.ok).toBe(true)
    expect(r.extended).toBe(true)
    expect((r as any).newClosing.toISOString()).toBe('2026-05-19T20:05:00.000Z')
  })
```

```sql
select
```

### `app/api/cron/close-lots/route.ts` (alternativa Vercel Cron)

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { createServerSupabase } from '@/lib/supabase/server'

export async function GET(req: NextRequest) {
  const auth = req.headers.get('authorization')
  if (auth !== `Bearer ${process.env.CRON_SECRET}`) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  }

  const supabase = await createServerSupabase()
  const { data: expired } = await supabase
    .from('lots').select('id')
    .in('status', ['active', 'extended'])
    .lte('closing_time', new Date().toISOString())

  for (const lot of expired ?? []) {
    await supabase.rpc('close_lot', { p_lot_id: lot.id })
  }

  return NextResponse.json({ closed: expired?.length ?? 0 })
}
```
